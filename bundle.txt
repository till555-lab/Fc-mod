@Mod(FemboyCompanion.MOD_ID)
public class FemboyCompanion {
    public static final String MOD_ID = "femboycompanion";
    public static final Logger LOGGER = LogManager.getLogger();
    public static final FemboyTrigger TRIGGER = new FemboyTrigger();

    public FemboyCompanion() {
        IEventBus bus = FMLJavaModLoadingContext.get().getModEventBus();
        ModEntities.register(bus);
        ModItems.register(bus);
        bus.addListener(this::onAttributes);
        bus.addListener(this::onCommonSetup);
        MinecraftForge.EVENT_BUS.addListener(this::onBiomeLoading);
        MinecraftForge.EVENT_BUS.addListener(this::onLivingDeath);
        LOGGER.info("Femboy Companion loaded");
    }

    private void onAttributes(EntityAttributeCreationEvent event) {
        event.put(ModEntities.FEMBOY.get(), FemboyEntity.createAttributes().build());
    }

    private void onCommonSetup(FMLCommonSetupEvent event) {
        event.enqueueWork(() -> {
            CriteriaTriggers.register(TRIGGER);
            EntitySpawnPlacementRegistry.register(ModEntities.FEMBOY.get(),
                    EntitySpawnPlacementRegistry.PlacementType.ON_GROUND,
                    Heightmap.Type.MOTION_BLOCKING_NO_LEAVES,
                    FemboyEntity::checkSpawnRules);
        });
    }

    private void onBiomeLoading(BiomeLoadingEvent event) {
        Biome.Category c = event.getCategory();
        if (c == Biome.Category.MUSHROOM || c == Biome.Category.OCEAN || c == Biome.Category.RIVER
                || c == Biome.Category.NETHER || c == Biome.Category.THEEND || c == Biome.Category.NONE) {
            return;
        }
        event.getSpawns().getSpawner(EntityClassification.CREATURE)
                .add(new MobSpawnInfo.Spawners(ModEntities.FEMBOY.get(), 6, 2, 4));
    }

    private void onLivingDeath(LivingDeathEvent event) {
        if (!(event.getEntity() instanceof FemboyEntity)) {
            return;
        }
        Entity killer = event.getSource().getEntity();
        if (!(killer instanceof PlayerEntity)) {
            return;
        }
        FemboyEntity dead = (FemboyEntity) event.getEntity();
        if (dead.level.isClientSide) {
            return;
        }
        AxisAlignedBB area = dead.getBoundingBox().inflate(16.0D);
        for (FemboyEntity witness : dead.level.getEntitiesOfClass(FemboyEntity.class, area)) {
            if (witness != dead) {
                witness.distrust(killer.getUUID(), dead.level.getGameTime() + 6000L);
            }
        }
    }
}

public class FemboyRenderer extends BipedRenderer<FemboyEntity, FemboyModel> {
    private static final ResourceLocation[][] TEXTURES =
            new ResourceLocation[2][FemboyEntity.COAT_COUNT];

    static {
        String[] families = {"white", "black"};
        for (int f = 0; f < 2; f++) {
            for (int c = 0; c < FemboyEntity.COAT_COUNT; c++) {
                TEXTURES[f][c] = new ResourceLocation(FemboyCompanion.MOD_ID,
                        "textures/entity/femboy_" + families[f] + "_" + c + ".png");
            }
        }
    }

    public FemboyRenderer(EntityRendererManager manager) {
        super(manager, new FemboyModel(), 0.5F);
    }

    @Override
    protected boolean shouldShowName(FemboyEntity entity) {
        if (entity.getAnger() > 0 && this.entityRenderDispatcher.distanceToSqr(entity) < 1024.0D) {
            return true;
        }
        return super.shouldShowName(entity);
    }

    @Override
    public ResourceLocation getTextureLocation(FemboyEntity entity) {
        int family = entity.getVariant() == FemboyEntity.VARIANT_BLACK ? 1 : 0;
        int coat = Math.floorMod(entity.getCoat(), FemboyEntity.COAT_COUNT);
        return TEXTURES[family][coat];
    }

    public void distrust(UUID id, long until) {

    private boolean isDistrusting(UUID id, long now) {
        Long until = this.distrusted.get(id);
        if (until == null) {
            return false;
        }
        if (now >= until) {
            this.distrusted.remove(id);
            this.trustPoints.remove(id);
            return false;
        }
        return true;
    }

    private static class GroupData implements ILivingEntityData {
        final int variant;

        GroupData(int variant) {
            this.variant = variant;
        }
    }

    private int rollVariant(IServerWorld world) {
        Biome biome = world.getBiome(this.blockPosition());
        Biome.Category category = biome.getBiomeCategory();
        ResourceLocation name = biome.getRegistryName();
        int whiteChance = 55;
        if (category == Biome.Category.ICY || category == Biome.Category.TAIGA) {
            whiteChance = 80;
        } else if (category == Biome.Category.SWAMP || category == Biome.Category.DESERT
                || category == Biome.Category.MESA
                || (name != null && name.getPath().equals("dark_forest"))) {
            whiteChance = 20;
        }
        return this.random.nextInt(100) < whiteChance ? VARIANT_WHITE : VARIANT_BLACK;
    }

    private int rollCoat() {
        return this.random.nextInt(3) == 0 ? 0 : 1 + this.random.nextInt(COAT_COUNT - 1);
    }

    @Override
    public ILivingEntityData finalizeSpawn(IServerWorld world, DifficultyInstance difficulty,
            SpawnReason reason, @Nullable ILivingEntityData data, @Nullable CompoundNBT tag) {
        int variant;
        if (data instanceof GroupData) {
            variant = ((GroupData) data).variant;
        } else {
            variant = this.rollVariant(world);
            data = new GroupData(variant);
        }
        this.setVariant(variant);
        this.setCoat(this.rollCoat());
        return super.finalizeSpawn(world, difficulty, reason, data, tag);
    }

    @Override
    public void aiStep() {
        super.aiStep();
        if (this.level.isClientSide) {
            return;
        }
        if (this.angerTicks > 0) {
            this.angerTicks--;
            if (this.angerTicks == 0) {
                this.calmDown(false);
            }
        }
        if (this.tickCount % 20 == 0) {
            PlayerEntity near = this.level.getNearestPlayer(this, 8.0D);
            if (near == null) {
                this.nearSeconds = 0;
            } else {
                this.nearSeconds++;
                if (this.nearSeconds >= 10) {
                    this.nearSeconds = 0;
                    this.fire(near, this.getVariant() == VARIANT_BLACK ? "meet_black" : "meet_white");
                }
            }
        }
    }

    private boolean isUpset() {
        return this.angerTicks > 0 || this.ignoring;
    }

    private void startAnger(PlayerEntity player) {
        this.angerTicks = 6000;
        this.ignoring = false;
        this.ignoreClicks = 0;
        this.calmPoints = 0;
        this.setItemSlot(EquipmentSlotType.MAINHAND, new ItemStack(Items.WOODEN_SWORD));
        this.syncAnger();
        this.say(player, "angry_start");
        this.fire(player, "sword");
        if (this.level.getDifficulty() != Difficulty.PEACEFUL && !player.isCreative() && !player.isSpectator()) {
            this.setTarget(player);
        }
    }

    private void calmDown(boolean full) {
        this.angerTicks = 0;
        this.ignoreClicks = 0;
        this.setTarget(null);
        this.setItemSlot(EquipmentSlotType.MAINHAND, ItemStack.EMPTY);
        if (full) {
            this.calmPoints = 0;
            this.ignoring = false;
            this.refusals = 0;
        } else {
            this.calmPoints = 50;
            this.ignoring = true;
        }
        this.syncAnger();
    }

    private static int calmPointsFor(ItemStack stack) {
        Item item = stack.getItem();
        if (ItemTags.FLOWERS.contains(item)) {
            return 10;
        }
        if (item == Items.WHEAT || item == Items.COOKIE) {
            return 5;
        }
        if (item == Items.CARROT) {
            return 6;
        }
        if (item == Items.APPLE) {
            return 8;
        }
        if (item == Items.BREAD) {
            return 15;
        }
        if (item == Items.COOKED_BEEF || item == Items.COOKED_PORKCHOP || item == Items.COOKED_CHICKEN
                || item == Items.COOKED_MUTTON || item == Items.COOKED_RABBIT || item == Items.COOKED_COD
                || item == Items.COOKED_SALMON) {
            return 20;
        }
        if (item == Items.GOLDEN_APPLE) {
            return 50;
        }
        if (item == Items.ENCHANTED_GOLDEN_APPLE) {
            return 100;
        }
        return 0;
    }

    @Override
    public ActionResultType mobInteract(PlayerEntity player, Hand hand) {
        if (hand != Hand.MAIN_HAND) {
            return super.mobInteract(player, hand);
        }
        ItemStack held = player.getItemInHand(hand);
        int points = held.isEmpty() ? 0 : calmPointsFor(held);
        if (!held.isEmpty() && points == 0) {
            return super.mobInteract(player, hand);
        }
        if (this.level.isClientSide) {
            return ActionResultType.SUCCESS;
        }
        if (this.tickCount - this.lastInteractTick < 10) {
            return ActionResultType.CONSUME;
        }
        this.lastInteractTick = this.tickCount;
        this.getLookControl().setLookAt(player, 30.0F, 30.0F);
        long now = this.level.getGameTime();
        UUID id = player.getUUID();

        if (points > 0) {
            if (this.isUpset()) {
                if (!player.abilities.instabuild) {
                    held.shrink(1);
                }
                this.calmPoints += points;
                if (this.calmPoints >= 100) {
                    this.calmDown(true);
                    this.say(player, "calm");
                    this.fire(player, "placated");
                } else {
                    this.syncAnger();
                }
                return ActionResultType.SUCCESS;
            }
            if (this.isDistrusting(id, now)) {
                if (!player.abilities.instabuild) {
                    held.shrink(1);
                }
                int total = this.trustPoints.getOrDefault(id, 0) + points;
                if (total >= 100) {
                    this.distrusted.remove(id);
                    this.trustPoints.remove(id);
                    this.say(player, "forgive");
                } else {
                    this.trustPoints.put(id, total);
                    player.displayClientMessage(
                            new TranslationTextComponent("message.femboycompanion.trust", total), true);
                }
                return ActionResultType.SUCCESS;
            }
            return super.mobInteract(player, hand);
        }

        if (now >= this.limitResetTime) {
            this.giftLimit = 2 + this.random.nextInt(9);
            this.giftsGiven = 0;
            this.refusals = 0;
            this.limitResetTime = now + 24000L;
        }
        if (this.isDistrusting(id, now)) {
            this.say(player, "distrust");
            return ActionResultType.CONSUME;
        }
        if (this.angerTicks > 0) {
            this.say(player, "angry");
            return ActionResultType.CONSUME;
        }
        if (this.ignoring) {
            this.ignoreClicks++;
            if (this.ignoreClicks >= 3) {
                this.startAnger(player);
            } else {
                this.say(player, "ignore");
            }
            return ActionResultType.SUCCESS;
        }
        if (this.giftsGiven < this.giftLimit) {
            this.giftsGiven++;
            this.giveGift(player);
            this.say(player, "gift");
            this.fire(player, "gift");
        } else {
            this.refusals++;
            if (this.refusals <= 3) {
                this.say(player, "refuse" + this.refusals);
            } else {
                this.startAnger(player);
            }
        }
        return ActionResultType.SUCCESS;
    }

    private void say(PlayerEntity player, String key) {
        player.displayClientMessage(new TranslationTextComponent("message.femboycompanion." + key), false);
    }

    private void giveGift(PlayerEntity player) {
        ItemStack stack = rollGift(this.random);
        ItemEntity item = new ItemEntity(this.level, this.getX(), this.getY() + 1.0D, this.getZ(), stack);
        Vector3d dir = player.position().subtract(this.position());
        Vector3d flat = new Vector3d(dir.x, 0.0D, dir.z).normalize().scale(0.25D);
        item.setDeltaMovement(flat.x, 0.2D, flat.z);
        item.setPickUpDelay(20);
        this.level.addFreshEntity(item);
    }

    private static ItemStack rollGift(Random random) {
        int r = random.nextInt(1000);
        if (r < 800) {
            Item[] common = {Items.DANDELION, Items.POPPY, Items.BLUE_ORCHID, Items.ALLIUM, Items.AZURE_BLUET,
                    Items.RED_TULIP, Items.ORANGE_TULIP, Items.WHITE_TULIP, Items.PINK_TULIP, Items.OXEYE_DAISY,
                    Items.CORNFLOWER, Items.LILY_OF_THE_VALLEY, Items.WHEAT_SEEDS, Items.PUMPKIN_SEEDS,
                    Items.MELON_SEEDS, Items.BEETROOT_SEEDS, Items.BONE, Items.COD, Items.SALMON, Items.STICK,
                    Items.APPLE, Items.CARROT, Items.POTATO};
            return pick(random, common, 3);
        }
        if (r < 950) {
            Item[] uncommon = {Items.BREAD, Items.COOKIE, Items.COOKED_COD, Items.COOKED_SALMON, Items.COAL,
                    Items.STRING, Items.SUGAR};
            return pick(random, uncommon, 2);
        }
        if (r < 990) {
            Item[] rare = {Items.IRON_INGOT, Items.GOLD_NUGGET, Items.LAPIS_LAZULI};
            return pick(random, rare, 2);
        }
        if (r < 999) {
            Item[] epic = {Items.GOLD_INGOT, Items.GOLDEN_CARROT, Items.EMERALD};
            return pick(random, epic, 1);
        }
        return new ItemStack(random.nextBoolean() ? Items.DIAMOND : Items.GOLDEN_APPLE);
    }

    private static ItemStack pick(Random random, Item[] items, int maxCount) {
        return new ItemStack(items[random.nextInt(items.length)], 1 + random.nextInt(maxCount));
    }

    @Override
    protected void dropCustomDeathLoot(DamageSource source, int looting, boolean recentlyHit) {
        super.dropCustomDeathLoot(source, looting, recentlyHit);
        int v = this.getVariant();
        double boost = v == VARIANT_BLACK ? 1.25D : 1.0D;
        if (this.random.nextDouble() < 0.20D * boost) {
            this.spawnAtLocation(new ItemStack(ModItems.ears(v)));
        }
        if (this.random.nextDouble() < 0.15D * boost) {
            this.spawnAtLocation(new ItemStack(ModItems.tail(v)));
        }
        if (this.random.nextDouble() < 0.20D * boost) {
            this.spawnAtLocation(new ItemStack(ModItems.paws(v)));
        }
    }

import os
import struct
import zlib

OUT = "src/main/resources/assets/femboycompanion/textures/item"


def write_png(path, rows):
    raw = b"".join(b"\x00" + b"".join(bytes(px) for px in row) for row in rows)

    def chunk(tag, data):
        body = struct.pack(">I", len(data)) + tag + data
        return body + struct.pack(">I", zlib.crc32(tag + data) & 0xFFFFFFFF)

    png = (b"\x89PNG\r\n\x1a\n"
           + chunk(b"IHDR", struct.pack(">IIBBBBB", 16, 16, 8, 6, 0, 0, 0))
           + chunk(b"IDAT", zlib.compress(raw, 9))
           + chunk(b"IEND", b""))
    with open(path, "wb") as f:
        f.write(png)


def main():
    os.makedirs(OUT, exist_ok=True)
    for fam, pal in PAL.items():
        for name, sprite in SPRITES.items():
            rows = []
            for line in sprite:
                assert len(line) == 16, name
                rows.append([pal[ch] + (255,) if ch != "." else (0, 0, 0, 0) for ch in line])
            write_png(os.path.join(OUT, "%s_%s.png" % (fam, name)), rows)
    print("item textures done")


main()

        "fur": (43, 43, 50), "paw": (74, 74, 88), "muz": (60, 60, 70),
        "ear": (138, 90, 112), "eye": (242, 242, 245), "nose": (232, 132, 159), "tip": (85, 85, 99),
    },
}


def speck_color(base, rnd):
    lum = 0.299 * base[0] + 0.587 * base[1] + 0.114 * base[2]
    k = 0.10 + rnd.random() * 0.14
    if lum >= 128:
        return tuple(int(c * (1 - k)) for c in base)
    return tuple(int(c + (255 - c) * k) for c in base)


def faces(u, v, w, h, d):
    return {
        "top": (u + d, v, w, d),
        "bottom": (u + d + w, v, w, d),
        "right": (u, v + d, d, h),
        "front": (u + d, v + d, w, h),
        "left": (u + d + w, v + d, d, h),
        "back": (u + 2 * d + w, v + d, w, h),
    }


def paint(kinds, u, v, w, h, d, fn):
    for name, (x0, y0, fw, fh) in faces(u, v, w, h, d).items():
        for ly in range(fh):
            for lx in range(fw):
                kinds[y0 + ly][x0 + lx] = fn(name, lx, ly)


def head_fn(name, lx, ly):
    if name == "front":
        if lx in (1, 6) and ly in (3, 4):
            return "eye"
        if lx in (3, 4) and ly == 5:
            return "nose"
        if ly >= 5 and 2 <= lx <= 5:
            return "muz"
    return "fur"


def body_fn(name, lx, ly):
    if name == "front" and 2 <= lx <= 5 and 2 <= ly <= 9:
        return "muz"
    return "fur"


def limb_fn(name, lx, ly):
    if name == "bottom":
        return "paw"
    if name != "top" and ly >= 9:
        return "paw"
    return "fur"


def ear_fn(name, lx, ly):
    if name == "front" and lx == 1 and 2 <= ly <= 10:
        return "ear"
    return "fur"


def tail_fn(name, lx, ly):
    return "tip" if name == "back" else "fur"


def build_kinds():
    kinds = [[None] * W for _ in range(H)]
    paint(kinds, 0, 0, 8, 8, 8, head_fn)
    paint(kinds, 16, 16, 8, 12, 4, body_fn)
    paint(kinds, 40, 16, 4, 12, 4, limb_fn)
    paint(kinds, 0, 16, 4, 12, 4, limb_fn)
    paint(kinds, 32, 0, 3, 12, 1, ear_fn)
    paint(kinds, 48, 0, 3, 3, 4, tail_fn)
    return kinds


def write_png(path, rows):
    raw = b"".join(b"\x00" + b"".join(bytes(px) for px in row) for row in rows)

    def chunk(tag, data):
        body = struct.pack(">I", len(data)) + tag + data
        return body + struct.pack(">I", zlib.crc32(tag + data) & 0xFFFFFFFF)

    png = (b"\x89PNG\r\n\x1a\n"
           + chunk(b"IHDR", struct.pack(">IIBBBBB", W, H, 8, 6, 0, 0, 0))
           + chunk(b"IDAT", zlib.compress(raw, 9))
           + chunk(b"IEND", b""))
    with open(path, "wb") as f:
        f.write(png)


def main():
    os.makedirs(OUT, exist_ok=True)
    kinds = build_kinds()
    for fam, pal in FAMILIES.items():
        for n, density in enumerate(DENSITY):
            rnd = random.Random(fam + "-" + str(n))
            rows = []
            for y in range(H):
                row = []
                for x in range(W):
                    kind = kinds[y][x]
                    if kind is None:
                        row.append((0, 0, 0, 0))
                        continue
                    color = pal[kind]
                    if kind == "fur" and rnd.random() < density:
                        color = speck_color(color, rnd)
                    row.append(color + (255,))
                rows.append(row)
            write_png(os.path.join(OUT, "femboy_%s_%d.png" % (fam, n)), rows)
    print("textures done")


main()
